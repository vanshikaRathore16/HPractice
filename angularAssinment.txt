Assignment 8:

Child to Parent using EventEmitter

child.component.ts

import { Component, Output, EventEmitter } from '@angular/core';

@Component({
  selector: 'app-child',
  template: `<button (click)="sendMsg()">Send to Parent</button>`
})
export class ChildComponent {
  @Output() messageEvent = new EventEmitter<string>();

  sendMsg() {
    this.messageEvent.emit('Hello Parent! From Child');
  }
}


parent.component.html

<app-child (messageEvent)="receiveMsg($event)"></app-child>
<p>{{ childMsg }}</p>


parent.component.ts

childMsg = '';

receiveMsg(msg: string) {
  this.childMsg = msg;
}


✅ Explanation:
Shows event binding and data emit from child → parent.

Assignment 9:

Build a CounterComponent (increase/decrease)

counter.component.ts

import { Component } from '@angular/core';

@Component({
  selector: 'app-counter',
  templateUrl: './counter.component.html'
})
export class CounterComponent {
  count = 0;

  inc() { this.count++; }
  dec() { this.count--; }
}


counter.component.html

<h3>Count: {{ count }}</h3>
<button (click)="inc()">+</button>
<button (click)="dec()">-</button>


✅ Explanation:
Used event binding with (click) to modify state.
